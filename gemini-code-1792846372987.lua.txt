function UIModule:Tween(instance, duration, properties, style, direction)
    style = style or Enum.EasingStyle.Quad
    direction = direction or Enum.EasingDirection.Out
    local info = TweenInfo.new(duration, style, direction)
    local tween = TweenService:Create(instance, info, properties)
    tween:Play()
    return tween
end

function UIModule:Create(className, properties, children)
    local instance = Instance.new(className)
    for k, v in pairs(properties or {}) do
        instance[k] = v
    end
    for _, child in ipairs(children or {}) do
        child.Parent = instance
    end
    return instance
end

function UIModule:Notify(title, text, duration)
    duration = duration or 3.5
    local host = PlayerGui:FindFirstChild("UINotifyHost")
    if not host then
        host = self:Create("ScreenGui", {
            Name = "UINotifyHost",
            Parent = PlayerGui,
            ResetOnSpawn = false,
            ZIndexBehavior = Enum.ZIndexBehavior.Sibling
        })
    end
    local container = host:FindFirstChild("Container")
    if not container then
        container = self:Create("Frame", {
            Name = "Container",
            Parent = host,
            BackgroundTransparency = 1,
            Position = UDim2.new(1, -20, 1, -20),
            Size = UDim2.new(0, 330, 1, -40),
            AnchorPoint = Vector2.new(1, 1)
        }, {
            self:Create("UIListLayout", {
                HorizontalAlignment = Enum.HorizontalAlignment.Right,
                VerticalAlignment = Enum.VerticalAlignment.Bottom,
                SortOrder = Enum.SortOrder.LayoutOrder,
                Padding = UDim.new(0, 10)
            })
        })
    end
    local notif = self:Create("Frame", {
        Parent = container,
        BackgroundColor3 = self.CurrentTheme.SecondaryBackground,
        Size = UDim2.new(1, 0, 0, 70),
        BackgroundTransparency = 1
    }, {
        self:Create("UICorner", { CornerRadius = UDim.new(0, 8) }),
        self:Create("UIStroke", { Color = self.CurrentTheme.AccentPrimary, Thickness = 1.5 }),
        self:Create("TextLabel", {
            BackgroundTransparency = 1,
            Position = UDim2.new(0, 15, 0, 8),
            Size = UDim2.new(1, -30, 0, 20),
            Font = Enum.Font.GothamBold,
            Text = title,
            TextColor3 = self.CurrentTheme.TextColor,
            TextSize = 13,
            TextXAlignment = Enum.TextXAlignment.Left
        }),
        self:Create("TextLabel", {
            BackgroundTransparency = 1,
            Position = UDim2.new(0, 15, 0, 28),

function UIModule:CreateTab(windowObj, tabName)
    local tabBtn = self:Create("TextButton", {
        Parent = windowObj.Sidebar,
        BackgroundColor3 = self.CurrentTheme.ElementBackground,
        Size = UDim2.new(1, 0, 0, 36),
        AutoButtonColor = false,
        Font = Enum.Font.GothamMedium,
        Text = "    " .. tabName,
        TextColor3 = self.CurrentTheme.SubTextColor,
        TextSize = 12,
        TextXAlignment = Enum.TextXAlignment.Left
    }, {
        self:Create("UICorner", { CornerRadius = UDim.new(0, 6) })
    })
    
    local indicator = self:Create("Frame", {
        Parent = tabBtn,
        BackgroundColor3 = self.CurrentTheme.AccentPrimary,
        BorderSizePixel = 0,
        Position = UDim2.new(0, 0, 0.15, 0),
        Size = UDim2.new(0, 3, 0.7, 0),
        BackgroundTransparency = 1
    }, {
        self:Create("UICorner", { CornerRadius = UDim.new(0, 2) })
    })
    
    local tabPage = self:Create("ScrollingFrame", {
        Name = tabName .. "Page",
        Parent = windowObj.Pages,
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 185, 0, 10),
        Size = UDim2.new(1, -195, 1, -20),
        CanvasSize = UDim2.new(0, 0, 0, 0),
        ScrollBarThickness = 3,
        Visible = false
    }, {
        self:Create("UIListLayout", {
            HorizontalAlignment = Enum.HorizontalAlignment.Center,
            SortOrder = Enum.SortOrder.LayoutOrder,
            Padding = UDim.new(0, 8)
        })
    })
    
    tabPage.UIListLayout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
        tabPage.CanvasSize = UDim2.new(0, 0, 0, tabPage.UIListLayout.AbsoluteContentSize.Y + 20)
    end)
    
    local function select()
        for _, p in ipairs(windowObj.Pages:GetChildren()) do p.Visible = false end
        for _, btn in ipairs(windowObj.Sidebar:GetChildren()) do
            if btn:IsA("TextButton") then
                self:Tween(btn, 0.2, {BackgroundColor3 = self.CurrentTheme.ElementBackground})
                btn.TextColor3 = self.CurrentTheme.SubTextColor
                if btn:FindFirstChild("Frame") then
                    self:Tween(btn.Frame, 0.2, {BackgroundTransparency = 1})
                end
            end
        end
        tabPage.Visible = true
        self:Tween(tabBtn, 0.2, {BackgroundColor3 = self.CurrentTheme.ElementHover})
        tabBtn.TextColor3 = self.CurrentTheme.TextColor
        self:Tween(indicator, 0.2, {BackgroundTransparency = 0})
    end
    
    tabBtn.MouseButton1Click:Connect(select)
    
    if windowObj.FirstTab then
        windowObj.FirstTab = false
        select()
    end
    
    return { Page = tabPage }
end

function UIModule:AddSearchBar(tabObj)
    local searchFrame = self:Create("Frame", {
        Parent = tabObj.Page,
        BackgroundColor3 = self.CurrentTheme.ElementBackground,
        Size = UDim2.new(1, -10, 0, 36)
    }, {
        self:Create("UICorner", { CornerRadius = UDim.new(0, 6) })
    })
    
    local input = self:Create("TextBox", {
        Parent = searchFrame,
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 12, 0, 0),
        Size = UDim2.new(1, -24, 1, 0),
        Font = Enum.Font.Gotham,
        PlaceholderText = "Search components...",
        PlaceholderColor3 = self.CurrentTheme.SubTextColor,
        Text = "",
        TextColor3 = self.CurrentTheme.TextColor,
        TextSize = 12,
        TextXAlignment = Enum.TextXAlignment.Left
    })
    
    input:GetPropertyChangedSignal("Text"):Connect(function()
        local query = input.Text:lower()
        for _, child in ipairs(tabObj.Page:GetChildren()) do
            if child:IsA("Frame") or child:IsA("TextButton") then
                if child ~= searchFrame then
                    if query == "" then
                        child.Visible = true
                    else
                        local label = child:FindFirstChildOfClass("TextLabel")
                        if label and label.Text:lower():find(query) then
                            child.Visible = true
                        else
                            child.Visible = false
                        end
                    end
                end
            end
        end
    end)
end

function UIModule:AddSlider(tabObj, text, min, max, default, callback)
    callback = callback or function() end
    default = math.clamp(default, min, max)
    
    local sliderFrame = self:Create("Frame", {
        Parent = tabObj.Page,
        BackgroundColor3 = self.CurrentTheme.ElementBackground,
        Size = UDim2.new(1, -10, 0, 50)
    }, {
        self:Create("UICorner", { CornerRadius = UDim.new(0, 6) }),
        self:Create("TextLabel", {
            BackgroundTransparency = 1,
            Position = UDim2.new(0, 12, 0, 6),
            Size = UDim2.new(1, -70, 0, 20),
            Font = Enum.Font.GothamMedium,
            Text = text,
            TextColor3 = self.CurrentTheme.TextColor,
            TextSize = 12,
            TextXAlignment = Enum.TextXAlignment.Left
        })
    })
    
    local valLabel = self:Create("TextLabel", {
        Parent = sliderFrame,
        BackgroundTransparency = 1,
        Position = UDim2.new(1, -65, 0, 6),
        Size = UDim2.new(0, 50, 0, 20),
        Font = Enum.Font.GothamBold,
        Text = tostring(default),
        TextColor3 = self.CurrentTheme.AccentPrimary,
        TextSize = 12,
        TextXAlignment = Enum.TextXAlignment.Right
    })
    
    local bar = self:Create("TextButton", {
        Parent = sliderFrame,
        BackgroundColor3 = Color3.fromRGB(45, 45, 60),
        Position = UDim2.new(0, 12, 0, 32),
        Size = UDim2.new(1, -24, 0, 6),
        AutoButtonColor = false,
        Text = ""
    }, {
        self:Create("UICorner", { CornerRadius = UDim.new(1, 0) })
    })
    
    local fill = self:Create("Frame", {
        Parent = bar,
        BackgroundColor3 = self.CurrentTheme.AccentPrimary,
        BorderSizePixel = 0,
        Size = UDim2.new((default - min) / (max - min), 0, 1, 0)
    }, {
        self:Create("UICorner", { CornerRadius = UDim.new(1, 0) })
    })
    
    local active = false
    local function update(input)
        local ratio = math.clamp((input.Position.X - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
        local val = math.floor(min + (max - min) * ratio)
        fill.Size = UDim2.new(ratio, 0, 1, 0)
        valLabel.Text = tostring(val)
        callback(val)
    end
    
    bar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            active = true
            update(input)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            active = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if active and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            update(input)
        end
    end)
end

        ClipsDescendants = true,
        Size = UDim2.new(1, -10, 0, 38)
    }, {

    })
    
    local mainBtn = self:Create("TextButton", {

        BackgroundTransparency = 1,

        Text = "    " .. text .. " : Select",
        TextColor3 = self.CurrentTheme.TextColor,
        TextSize = 12,
        TextXAlignment = Enum.TextXAlignment.Left
    })
    
    local container = self:Create("ScrollingFrame", {
        Parent = dropFrame,
        BackgroundTransparency = 1,
        Position = UDim2.new(0, 0, 0, 38),
        Size = UDim2.new(1, 0, 0, 100),
        CanvasSize = UDim2.new(0, 0, 0, #list * 28),

            HorizontalAlignment = Enum.HorizontalAlignment.Center,
            SortOrder = Enum.SortOrder.LayoutOrder,

        end)

function UIModule:AddKeybind(tabObj, text, defaultKey, callback)
    callback = callback or function() end
    local boundKey = defaultKey or Enum.KeyCode.E
    local listening = false
    
    local kbFrame = self:Create("Frame", {
        Parent = tabObj.Page,
        BackgroundColor3 = self.CurrentTheme.ElementBackground,
        Size = UDim2.new(1, -10, 0, 38)
    }, {
        self:Create("UICorner", { CornerRadius = UDim.new(0, 6) }),
        self:Create("TextLabel", {
            BackgroundTransparency = 1,
            Position = UDim2.new(0, 12, 0, 0),
            Size = UDim2.new(1, -90, 1, 0),
            Font = Enum.Font.GothamMedium,
            Text = text,
            TextColor3 = self.CurrentTheme.TextColor,
            TextSize = 12,
            TextXAlignment = Enum.TextXAlignment.Left
        })
    })
    
    local bindBtn = self:Create("TextButton", {
        Parent = kbFrame,
        BackgroundColor3 = Color3.fromRGB(40, 40, 55),
        Position = UDim2.new(1, -75, 0.5, -12),
        Size = UDim2.new(0, 65, 0, 24),
        AutoButtonColor = false,
        Font = Enum.Font.GothamBold,
        Text = boundKey.Name,
        TextColor3 = self.CurrentTheme.AccentPrimary,
        TextSize = 11
    }, {
        self:Create("UICorner", { CornerRadius = UDim.new(0, 4) })
    })
    
    bindBtn.MouseButton1Click:Connect(function()
        listening = true
        bindBtn.Text = "..."
    end)
    
    UserInputService.InputBegan:Connect(function(input, processed)
        if listening then
            if input.UserInputType == Enum.UserInputType.Keyboard then
                boundKey = input.KeyCode
                bindBtn.Text = boundKey.Name
                listening = false
            end
        elseif not processed and input.KeyCode == boundKey then
            callback(boundKey)
        end
    end)
end

function UIModule:AddDataTable(tabObj, headers, rows)
    headers = headers or {"Item", "Status", "Value"}
    rows = rows or {}
    
    local tableFrame = self:Create("Frame", {
        Parent = tabObj.Page,
        BackgroundColor3 = self.CurrentTheme.ElementBackground,
        Size = UDim2.new(1, -10, 0, 30 + (#rows * 24))
    }, {
        self:Create("UICorner", { CornerRadius = UDim.new(0, 6) })
    })
    
    local headerFrame = self:Create("Frame", {
        Parent = tableFrame,
        BackgroundColor3 = self.CurrentTheme.SecondaryBackground,
        Size = UDim2.new(1, 0, 0, 26)
    }, {
        self:Create("UICorner", { CornerRadius = UDim.new(0, 6) })
    })
    
    local colWidth = 1 / #headers
    for i, hText in ipairs(headers) do
        self:Create("TextLabel", {
            Parent = headerFrame,
            BackgroundTransparency = 1,
            Position = UDim2.new((i - 1) * colWidth, 10, 0, 0),
            Size = UDim2.new(colWidth, -10, 1, 0),
            Font = Enum.Font.GothamBold,
            Text = hText,
            TextColor3 = self.CurrentTheme.TextColor,
            TextSize = 11,
            TextXAlignment = Enum.TextXAlignment.Left
        })
    end
    
    for rowIndex, rowData in ipairs(rows) do
        local rowFrame = self:Create("Frame", {
            Parent = tableFrame,
            BackgroundTransparency = 1,
            Position = UDim2.new(0, 0, 0, 26 + ((rowIndex - 1) * 24)),
            Size = UDim2.new(1, 0, 0, 24)
        })
        
        for colIndex, cellValue in ipairs(rowData) do
            self:Create("TextLabel", {
                Parent = rowFrame,
                BackgroundTransparency = 1,
                Position = UDim2.new((colIndex - 1) * colWidth, 10, 0, 0),
                Size = UDim2.new(colWidth, -10, 1, 0),
                Font = Enum.Font.Gotham,
                Text = tostring(cellValue),
                TextColor3 = self.CurrentTheme.SubTextColor,
                TextSize = 11,
                TextXAlignment = Enum.TextXAlignment.Left
            })
        end
    end
end
