Grid.CellPadding = UDim2.new(0, 8, 0, 8)
Grid.Parent = Scroll

-- -------------------------------------------------------------
-- Ultra-Fast Rendering Engine (Chunked Batch Processing)
-- -------------------------------------------------------------
-- เตรียม Index รายการล่วงหน้าเพื่อเพิ่มความเร็ว
local SortedIcons = {}
for name, id in pairs(BuiltInIcons) do
    table.insert(SortedIcons, {name = name, lowerName = name:lower(), id = id})
end
table.sort(SortedIcons, function(a, b) return a.name < b.name end)

local renderThread = nil
local activeCards = {}

local function clearGrid()
    for _, child in ipairs(Scroll:GetChildren()) do
        if child:IsA("GuiObject") then
            child:Destroy()
        end
    end
    activeCards = {}
end

local function render()
    if renderThread then
        task.cancel(renderThread)
        renderThread = nil
    end

    clearGrid()
    local q = Search.Text:lower():gsub("%%", "%%%%")

    renderThread = task.spawn(function()
        local count = 0
        local batchSize = 30 -- สร้างผลลัพธ์ทีละกลุ่ม เพื่อไม่ให้ UI กระตุก

        for _, item in ipairs(SortedIcons) do
            if q == "" or item.lowerName:find(q, 1, true) then
                count = count + 1

                local Card = Instance.new("TextButton")
                Card.Name = item.name
                Card.Text = ""
                Card.BackgroundColor3 = Color3.fromRGB(24, 28, 40)
                Card.AutoButtonColor = false
                Card.Parent = Scroll

                local CardCorner = Instance.new("UICorner")
                CardCorner.CornerRadius = UDim.new(0, 8)
                CardCorner.Parent = Card

                local CardStroke = Instance.new("UIStroke")
                CardStroke.Thickness = 1
                CardStroke.Color = Color3.fromRGB(38, 45, 62)
                CardStroke.Parent = Card

                local Img = Instance.new("ImageLabel")
                Img.Size = UDim2.new(0, 32, 0, 32)
                Img.Position = UDim2.new(0.5, -16, 0.18, 0)
                Img.Image = item.id
                Img.BackgroundTransparency = 1
                Img.Parent = Card

                local Lbl = Instance.new("TextLabel")
                Lbl.Size = UDim2.new(1, -6, 0, 26)
                Lbl.Position = UDim2.new(0, 3, 0.62, 0)
                Lbl.Text = item.name
                Lbl.TextColor3 = Color3.fromRGB(180, 190, 210)
                Lbl.BackgroundTransparency = 1
                Lbl.Font = Enum.Font.Gotham
                Lbl.TextSize = 10
                Lbl.TextWrapped = true
                Lbl.Parent = Card

                -- Hover Effect
                Card.MouseEnter:Connect(function()
                    Card.BackgroundColor3 = Color3.fromRGB(32, 38, 55)
                    CardStroke.Color = Color3.fromRGB(0, 210, 255)
                end)
                Card.MouseLeave:Connect(function()
                    Card.BackgroundColor3 = Color3.fromRGB(24, 28, 40)
                    CardStroke.Color = Color3.fromRGB(38, 45, 62)
                end)

                -- Click to Copy
                Card.MouseButton1Click:Connect(function()
                    if setclipboard then
                        setclipboard(item.id)
                    end
                    Card.BackgroundColor3 = Color3.fromRGB(0, 180, 120)
                    Lbl.Text = "COPIED!"
                    Lbl.TextColor3 = Color3.fromRGB(255, 255, 255)
                    task.wait(0.4)
                    Card.BackgroundColor3 = Color3.fromRGB(24, 28, 40)
                    Lbl.Text = item.name
                    Lbl.TextColor3 = Color3.fromRGB(180, 190, 210)
                end)

                if count % batchSize == 0 then
                    Scroll.CanvasSize = UDim2.new(0, 0, 0, Grid.AbsoluteContentSize.Y + 10)
                    task.wait() -- พัก Frame เพื่อให้ลื่นไหลและโหลดเร็วที่สุด
                end
            end
        end
        Scroll.CanvasSize = UDim2.new(0, 0, 0, Grid.AbsoluteContentSize.Y + 10)
    end)
end

Search:GetPropertyChangedSignal("Text"):Connect(render)
render()

-- Smooth Draggable System
local dragging, dragInput, dragStart, startPos
Header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

Header.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UIS.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)
